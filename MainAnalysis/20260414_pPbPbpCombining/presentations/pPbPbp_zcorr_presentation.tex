\begin{frame}[t]{Z correction: pPb$\to$Pbp}
  \small
  \textbf{Method:} iterative 3D ($p_T$, $y_\text{CM}$, $\phi$) reweighting of pPb Z kinematics to match Pbp\par
  \vspace{0.3em}
  \textbf{Iterations:} 3 (corrections converge to $\sim$1.0 by iter 3)\par
  \vspace{0.3em}
  \textbf{pPb data:} Z correction applied multiplicatively to eventWeight\par
  \textbf{Pbp data:} unchanged (target distribution)\par
  \vspace{0.3em}
  \textbf{Binning:} $p_T$: 50 log bins [0.5, 100]~GeV;\ $y_\text{CM}$: 12 bins [$-3.5$, 2.5];\ $\phi$: 12 bins [0, $2\pi$]\par
  \vspace{0.3em}
  \textbf{Empty-bin protection:} bins with $<1$ event in denominator set to 1.0\par
  \vspace{0.5em}
  {\scriptsize
  \begin{tabular}{l r r}
              & pPb NZ    & Pbp NZ \\

  +VZ         & 7027      & 7035 \\

  +VZ+Z+ZSF+trk     & 5937      & 5943 \\
  +VZ+Z+ZSF+trk+res & 5937      & 5943 \\
  \end{tabular}}
\end{frame}
