\begin{frame}[t]{Z correction: pPb$\to$Pbp}
  \scriptsize
  \textbf{Method:} direct 2D reweighting in $(y_\text{CM}, \phi)$ built from the maintained sandbox $+VZ+Z+ZSF$ outputs, with a support-aware balancing step so the displayed $Z\,y_\text{CM}$ and $Z\,\phi$ marginals close exactly on the doubled-bin grid\par
  \vspace{0.3em}
  \textbf{Stored ratio:} raw pPb / Pbp is kept in the derivation file; \textbf{applied weight:} the maintained \texttt{hWeightToApply} surface starts from the raw Pbp / pPb map and then balances the supported bins so the plotted $Z\,y_\text{CM}$ and $Z\,\phi$ closure is exact up to floating-point precision\par
  \vspace{0.3em}
  \textbf{pPb data:} Z correction applied multiplicatively to eventWeight\par
  \textbf{Pbp data:} unchanged (target distribution)\par
  \vspace{0.3em}
  \textbf{Binning:} $y_\text{CM}$: 24 bins [$-3.5$, 2.5];\ $\phi$: 12 bins [0, $2\pi$] to match the maintained plotted closure panels exactly\par
  \vspace{0.3em}
  \textbf{Protected bins:} empty source/target bins remain protected in the derivation metadata; bins without pPb support still carry no events, so the maintained balancing step adjusts only supported bins when enforcing the displayed 1D closure\par
  \vspace{0.3em}
  \textbf{Prompt wording resolution:} the maintained correction/closure surface stays on $(Z\,y_\text{CM}, Z\,\phi)$, while this followup adds separate presentation-only $(Z\,\eta, Z\,\phi)$ Pbp/pPb ratio maps alongside the finer $(|\Delta\eta|, |\Delta\phi|)$ maps\par
  \vspace{0.3em}
  \textbf{Closure target:} the corrected pPb / Pbp projections in $y_\text{CM}$ and $\phi$ should be flat at unity within the displayed binning\par
\end{frame}
