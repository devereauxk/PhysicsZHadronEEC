% ---- Study 0: KS-like test slide ----
\begin{frame}[t]{Study 0: Covariance-aware KS-like test --- pPb vs Pbp (nominal)}
  \vspace{0.3em}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \resizebox{\linewidth}{!}{\input{\bno/pPbPbp_covKS.tex}}
    \end{column}
    \begin{column}{0.43\textwidth}
      \small
      Covariance-aware cumulative shape statistic:\\[0.3em]
      $T_{\rm max} = \max_k |Z_k|$,\quad $Z_k = S_k / \sqrt{V_{kk}}$\\[0.2em]
      where $S_k = \sum_{i\leq k} \Delta_i$ (cumulative difference)\\
      and $V_{kk}$ = cumulative variance from full $C^{\rm tot} = C^{pPb} + C^{Pbp}$.\\[0.4em]
      $p$-value from 100\,000 toy MC replicas drawn from\\
      $\Delta \sim \mathcal{N}(0, C^{\rm tot})$.\\[0.3em]
      Full jackknife covariance matrices used.\\[0.2em]
      Not a replacement for the $\chi^2$ test;\\
      sensitive to shape offsets that persist across bins.
    \end{column}
  \end{columns}
\end{frame}

% ---- Study 0: standard KS test slide ----
\begin{frame}[t]{Study 0: Standard KS test --- pPb vs Pbp (nominal)}
  \small
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \resizebox{\linewidth}{!}{\input{\bno/pPbPbp_ks_bare.tex}}
    \end{column}
    \begin{column}{0.43\textwidth}
      \footnotesize
      \texttt{ROOT::Math::GoFTest}\\
      \texttt{::KolmogorovSmirnov2SamplesTest}\\[0.3em]
      Input: the 12 result-histogram bin values per sample\\
      treated as two independent data sets.\\[0.2em]
      $D_{\rm KS} = \max_x |F_{\rm pPb}(x) - F_{\rm Pbp}(x)|$\\[0.1em]
      where $F$ is the empirical CDF over sorted bin values.\\
      Negative bins handled natively — no shift needed.\\[0.2em]
      $p$-value uses the asymptotic Kolmogorov distribution\\
      with $n_{\rm eff} = n_1 n_2/(n_1{+}n_2) = 6$\\
      (i.e.\ based on $N_{\rm bins}=12$ per sample,\\
      not on the much larger $N_Z$).\\[0.2em]
      Compare to the covariance-aware KS-like test\\
      (previous slide) which sums bins in $\Delta\eta$ order\\
      and uses the full jackknife covariance.
    \end{column}
  \end{columns}
\end{frame}

\sectionframe{Study 9: Bin-by-bin ME correction to Pbp}{%
  Pbp ME replaced bin-by-bin with the pPb ME: $ME_{\rm Pbp}^{\rm corr}[i,j] = ME_{\rm pPb}[i,j]$.\\[0.3em]
  This forces the ME ratio $ME_{\rm Pbp}^{\rm corr}/ME_{\rm pPb}$ to unity in every 2D bin,
  and therefore flat in both $\Delta\eta$ and $\Delta\phi$ projections.\\[0.3em]
  Corrected Pbp result $= SE_{\rm Pbp} - ME_{\rm pPb}$; residual disagreement with pPb reflects
